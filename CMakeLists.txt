option(ZPLANE_VERIFY_MORPHOSIS_ORACLE "Build independent pinned Morphosis DSP oracle (development QA only)" OFF)
if(ZPLANE_VERIFY_MORPHOSIS_ORACLE AND BUILD_TESTING)
 set(_morphosis_sha "d271a8a8c6cf8bb5524e588c3497f4b5544a22cd")
 set(_morphosis_dir "${CMAKE_BINARY_DIR}/reference_morphosis")
 file(MAKE_DIRECTORY "${_morphosis_dir}")
 # Deliberately share the 125A embedded CubeData.h with the independent
 # reference DSP. Fetching a second CubeData.h would redefine the same
 # constexpr record types in the oracle translation unit.
 foreach(_morphosis_file IN ITEMS MorphosisDSP.h MorphosisDSP.cpp MorphosisSequencer.h)
  file(DOWNLOAD "https://raw.githubusercontent.com/limitsurface/morphosis_vst/${_morphosis_sha}/src/${_morphosis_file}"
   "${_morphosis_dir}/${_morphosis_file}" STATUS _morphosis_status TLS_VERIFY ON TIMEOUT 60)
  list(GET _morphosis_status 0 _morphosis_result)
  if(NOT _morphosis_result EQUAL 0)
   message(FATAL_ERROR "Pinned Morpheus oracle unavailable: ${_morphosis_file}: ${_morphosis_status}")
  endif()
 endforeach()
 add_executable(zplane_morpheus_oracle_tests
   tests/MorpheusOracleTests.cpp "${_morphosis_dir}/MorphosisDSP.cpp")

 target_include_directories(zplane_morpheus_oracle_tests PRIVATE "${_morphosis_dir}")

 if(ZPLANE_WITH_X3_RESEARCH)
  set(_research_zip "${CMAKE_BINARY_DIR}/x3-research.zip")
  file(DOWNLOAD
   "https://raw.githubusercontent.com/noneofyourbisness/EMU-Emulator-X3-ZPlaneFilter-Research/main/emulator-x3-filter-source.zip"
   "${_research_zip}"
   EXPECTED_HASH SHA256=91062ac20281ef0892dc9bc9696df76106c259f60d4fa7fcff4175dc5c7f3b88
   TLS_VERIFY ON STATUS _research_status)
  list(GET _research_status 0 _research_code)
  if(NOT _research_code EQUAL 0)
   message(FATAL_ERROR "Research archive download failed: ${_research_status}")
  endif()
  file(ARCHIVE_EXTRACT INPUT "${_research_zip}" DESTINATION "${CMAKE_BINARY_DIR}/third_party_research")
  target_include_directories(ZPlaneFX PRIVATE "${CMAKE_BINARY_DIR}/third_party_research/emulator-x3-filters")
  target_compile_definitions(ZPlaneFX PRIVATE ZPLANE_WITH_X3_RESEARCH=1)
  # Embed the three byte-exact reconstructed X3 filter banks directly into
  # the PE: no end-user Bank directory, installer step or environment needed.
  # CI reconstructs/validates the pinned source-bank hashes before build.
  set(_x3_assets "${CMAKE_BINARY_DIR}/third_party_research/emulator-x3-filters")
  file(TO_CMAKE_PATH "${_x3_assets}/factory-bank.bin" _x3_factory)
  file(TO_CMAKE_PATH "${_x3_assets}/standard-bank.bin" _x3_standard)
  file(TO_CMAKE_PATH "${_x3_assets}/morph-support.bin" _x3_support)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ZPlaneEmbeddedX3.rc"
       "2001 RCDATA \"${_x3_factory}\"\n"
       "2002 RCDATA \"${_x3_standard}\"\n"
       "2003 RCDATA \"${_x3_support}\"\n")
  target_sources(ZPlaneFX PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/ZPlaneEmbeddedX3.rc")
  if(MSVC)
   target_compile_options(ZPlaneFX PRIVATE /fp:strict)
  endif()
