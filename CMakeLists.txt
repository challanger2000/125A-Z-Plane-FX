cmake_minimum_required(VERSION 3.25)
project(ZPlaneFX VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(zplane_core INTERFACE)
target_include_directories(zplane_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(zplane_core INTERFACE cxx_std_23)
if(BUILD_TESTING)
 add_executable(zplane_core_tests tests/FilterCoreTests.cpp)
 target_link_libraries(zplane_core_tests PRIVATE zplane_core)
 if(MSVC)
  target_compile_options(zplane_core_tests PRIVATE /W4 /WX)
 endif()
 add_test(NAME zplane_core_tests COMMAND zplane_core_tests)
 add_executable(zplane_automation_tests tests/AutomationTimelineTests.cpp)
 target_link_libraries(zplane_automation_tests PRIVATE zplane_core)
 if(MSVC)
  target_compile_options(zplane_automation_tests PRIVATE /W4 /WX)
 endif()
 add_test(NAME zplane_automation_tests COMMAND zplane_automation_tests)
endif()
option(ZPLANE_BUILD_VST3 "Build Windows-only VST3" ON)
if(ZPLANE_BUILD_VST3)
 if(NOT WIN32)
  message(FATAL_ERROR "125A Z-Plane FX only supports Windows x64")
 endif()
 if(NOT DEFINED VST3_SDK_ROOT OR NOT EXISTS "${VST3_SDK_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "VST3_SDK_ROOT must point to pinned recursive SDK")
 endif()
 set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
 set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
 set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
 set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)
 add_subdirectory("${VST3_SDK_ROOT}" "${CMAKE_BINARY_DIR}/sdk")
 smtg_add_vst3plugin(ZPlaneFX src/Vst3.cpp)
 target_link_libraries(ZPlaneFX PRIVATE sdk zplane_core)
 target_compile_features(ZPlaneFX PRIVATE cxx_std_23)
 if(BUILD_TESTING)
  add_executable(zplane_native_host_tests tests/NativeHostTests.cpp)
  target_link_libraries(zplane_native_host_tests PRIVATE sdk_hosting sdk_common base pluginterfaces)
  target_compile_features(zplane_native_host_tests PRIVATE cxx_std_23)
  if(MSVC)
   target_compile_options(zplane_native_host_tests PRIVATE /W4 /WX)
  endif()
 endif()
 set_target_properties(ZPlaneFX PROPERTIES OUTPUT_NAME "125A Z-Plane FX")
endif()
